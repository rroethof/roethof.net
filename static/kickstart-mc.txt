#!/bin/bash

UUID=$(cat /sys/class/dmi/id/product_uuid 2>/dev/null | tr -d '-')

if [ -z "$UUID" ] || \
   [ "$UUID" = "00000000000000000000000000000000" ]; then
    UUID=$(uuidgen | tr -d '-')
fi

SHORT_UUID="${UUID:0:8}"

VG_OS="vg_os_${SHORT_UUID}"
VG_MC="vg_mc_${SHORT_UUID}"


cat > /tmp/vgname.ks <<EOF
volgroup ${VG_OS} pv.01 --pesize=4096
volgroup ${VG_MC} pv.02 --pesize=4096
EOF


cat > /tmp/lvs.ks <<EOF

#=======================================================
# OPERATING SYSTEM
#=======================================================

logvol swap \
    --vgname=${VG_OS} \
    --name=lv_swap \
    --size=4096

logvol / \
    --vgname=${VG_OS} \
    --name=lv_root \
    --size=20480 \
    --fstype="xfs"

logvol /tmp \
    --vgname=${VG_OS} \
    --name=lv_tmp \
    --size=2048 \
    --fstype="xfs" \
    --fsoptions="nodev,nosuid"

logvol /var \
    --vgname=${VG_OS} \
    --name=lv_var \
    --size=8192 \
    --fstype="xfs" \
    --fsoptions="nodev"

logvol /var/log \
    --vgname=${VG_OS} \
    --name=lv_var_log \
    --size=4096 \
    --fstype="xfs" \
    --fsoptions="nodev,nosuid"

logvol /var/log/audit \
    --vgname=${VG_OS} \
    --name=lv_var_log_audit \
    --size=4096 \
    --fstype="xfs" \
    --fsoptions="nodev,nosuid"

logvol /var/tmp \
    --vgname=${VG_OS} \
    --name=lv_var_tmp \
    --size=2048 \
    --fstype="xfs" \
    --fsoptions="nodev,nosuid"


#=======================================================
# MINECRAFT
#=======================================================

logvol /srv \
    --vgname=${VG_MC} \
    --name=lv_minecraft \
    --size=1 \
    --grow \
    --fstype="xfs" \
    --fsoptions="nodev"

EOF


%end

#!/bin/bash

set -x

trap 'echo "FOUT op regel $LINENO — zie /var/log/kickstart.log"' ERR


#=======================================================
# VERSION PINNING
#=======================================================

MC_VERSION="26.2"

JAVA_VERSION="25"

FABRIC_PROXY_LITE_VERSION="2.12.0"

VIAVERSION_VERSION="5.10.0"


#=======================================================
# DIRECTORIES
#=======================================================

MINECRAFT_ROOT="/srv/minecraft"

VELOCITY_ROOT="${MINECRAFT_ROOT}/velocity"

HUB_ROOT="${MINECRAFT_ROOT}/hub"

BACKUP_ROOT="${MINECRAFT_ROOT}/backups"


#=======================================================
# USERS
#=======================================================

useradd \
    --system \
    --home-dir "${VELOCITY_ROOT}" \
    --shell /sbin/nologin \
    minecraft-proxy \
    || true

    --system \
    --home-dir "${HUB_ROOT}" \
    --shell /sbin/nologin \
    minecraft-hub \
    || true

mkdir -p "${HUB_ROOT}/mods"

mkdir -p "${HUB_ROOT}/config"

mkdir -p "${HUB_ROOT}/logs"

mkdir -p "${BACKUP_ROOT}"


chown -R minecraft-proxy:minecraft-proxy \
    "${VELOCITY_ROOT}"

chown -R minecraft-hub:minecraft-hub \
    "${HUB_ROOT}"

chown -R minecraft-hub:minecraft-hub \
    "${BACKUP_ROOT}"


#=======================================================
# JAVA VALIDATION
#=======================================================

echo "======================================================="
echo " Java"
echo "======================================================="

java -version


JAVA_MAJOR=$(
    java -version 2>&1 \
    | awk -F '"' '/version/ {print $2}' \
    | cut -d. -f1
)

    echo "ERROR: Java 25 required"

    exit 1

fi

# SSH HARDENING
#=======================================================

PermitRootLogin no

PasswordAuthentication no

KbdInteractiveAuthentication no

PubkeyAuthentication yes

AuthenticationMethods publickey

MaxAuthTries 3

AllowAgentForwarding no

AllowTcpForwarding no

X11Forwarding no

PermitTunnel no

ClientAliveInterval 300

ClientAliveCountMax 0

EOF


#=======================================================
# SSH KEY
#=======================================================

mkdir -p /home/rroethof/.ssh

chmod 700 /home/rroethof/.ssh


cat > /home/rroethof/.ssh/authorized_keys <<'EOF'
ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIMELqQdsIVHY4dU/rwGVkgAb1MV+TQM9rFzYDhweWafB rroethof@kickstart
EOF


chmod 600 \
    /home/rroethof/.ssh/authorized_keys


chown -R rroethof:rroethof \
    /home/rroethof/.ssh


#=======================================================
# SUDO
#=======================================================

cat > /etc/sudoers.d/rroethof <<'EOF'
rroethof ALL=(ALL) ALL
EOF

    /etc/sudoers.d/rroethof

# FIREWALL
#=======================================================

    --add-service=ssh

    --add-port=25565/tcp

# FABRIC INSTALLER
#=======================================================
#
# We retrieve the current Fabric installer from Maven.
#
# The installer itself determines the appropriate loader
# for Minecraft 26.2.
#=======================================================

echo "======================================================="
echo " Installing Fabric ${MC_VERSION}"
echo "======================================================="


cd "${HUB_ROOT}"


FABRIC_INSTALLER_VERSION=$(
    curl -fsSL \
        "https://meta.fabricmc.net/v2/versions/installer" \
    | jq -r '[.[] | select(.stable == true)][0].version'
)


if [ -z "${FABRIC_INSTALLER_VERSION}" ] || \
   [ "${FABRIC_INSTALLER_VERSION}" = "null" ]; then

    echo "ERROR: unable to determine Fabric installer"

    exit 1

fi


curl -fL \
    -o fabric-installer.jar \
    "https://maven.fabricmc.net/net/fabricmc/fabric-installer/${FABRIC_INSTALLER_VERSION}/fabric-installer-${FABRIC_INSTALLER_VERSION}.jar"


chown minecraft-hub:minecraft-hub \
    fabric-installer.jar


#=======================================================
# INSTALL FABRIC
#=======================================================

su -s /bin/bash minecraft-hub -c \
    "cd '${HUB_ROOT}' && \
     ${JAVA_BIN:-/usr/bin/java} \
     -jar fabric-installer.jar \
     server \
     -mcversion '${MC_VERSION}' \
     -downloadMinecraft"


#=======================================================
# EULA
#=======================================================

    > "${HUB_ROOT}/eula.txt"


#=======================================================
# FABRIC API
#=======================================================
#
# Current compatible Fabric API is discovered through
# Modrinth.
#
# This avoids hardcoding a stale API build.
#=======================================================

echo "======================================================="
echo " Installing Fabric API"
echo "======================================================="


FABRIC_API_URL=$(
    curl -fsSL \
        "https://api.modrinth.com/v2/project/fabric-api/version?game_versions=%5B%22${MC_VERSION}%22%5D&loaders=%5B%22fabric%22%5D&limit=20" \
    | jq -r '
        map(select(.version_type == "release"))
        | .[0].files[]
        | select(.primary == true)
        | .url
    ' \
    | head -n1
)


if [ -z "${FABRIC_API_URL}" ] || \
   [ "${FABRIC_API_URL}" = "null" ]; then

    echo "ERROR: no release Fabric API found for ${MC_VERSION}"

    exit 1

fi


curl -fL \
    -o "${HUB_ROOT}/mods/fabric-api.jar" \
    "${FABRIC_API_URL}"

# Current known release:
#

echo "======================================================="


FABRIC_PROXY_URL=$(
    curl -fsSL \
        "https://api.modrinth.com/v2/project/fabricproxy-lite/version?loaders=%5B%22fabric%22%5D&game_versions=%5B%22${MC_VERSION}%22%5D&limit=20" \
    | jq -r '
        map(select(.version_type == "release"))
        | .[0].files[]
        | select(.primary == true)
        | .url
    ' \
    | head -n1

if [ -z "${FABRIC_PROXY_URL}" ] || \

    echo "ERROR: no FabricProxy-Lite release found"

    -o "${HUB_ROOT}/mods/FabricProxy-Lite.jar" \

    curl -fsSL \
        "https://api.modrinth.com/v2/project/viafabric/version?loaders=%5B%22fabric%22%5D&game_versions=%5B%22${MC_VERSION}%22%5D&limit=20" \
    | jq -r '
        map(select(.version_type == "release" or .version_type == "beta"))
        | .[0].files[]
        | select(.primary == true)
        | .url
    ' \
    | head -n1
)


if [ -n "${VIAFABRIC_URL}" ] && \
   [ "${VIAFABRIC_URL}" != "null" ]; then

    curl -fL \
        -o "${HUB_ROOT}/mods/ViaFabric.jar" \

# START FABRIC
#=======================================================

echo "======================================================="
echo " Starting Fabric Hub"
echo "======================================================="


systemctl start minecraft-hub.service


#=======================================================
# WAIT FOR FABRIC
#=======================================================

for i in {1..90}; do

    if ss -lnt | grep -q '127.0.0.1:25566 '; then

        echo "Fabric Hub is listening."

        break

    fi

    sleep 2

done


#=======================================================
# START VELOCITY
#=======================================================

echo "======================================================="
echo " Starting Velocity"
echo "======================================================="


systemctl start velocity.service


#=======================================================
# NODE INFORMATION
#=======================================================

cat > /etc/minecraft-node.conf <<EOF

NODE_ROLE=hub

MINECRAFT_VERSION=${MC_VERSION}

JAVA_VERSION=${JAVA_VERSION}

VELOCITY_VERSION=${VELOCITY_VERSION}

FABRIC_PROXY_LITE_VERSION=${FABRIC_PROXY_LITE_VERSION}

VIAVERSION_VERSION=${VIAVERSION_VERSION}

VELOCITY_PORT=25565

FABRIC_PORT=25566

FABRIC_BIND=127.0.0.1

VELOCITY_ROOT=${VELOCITY_ROOT}

HUB_ROOT=${HUB_ROOT}

EOF


#=======================================================
# FINAL STATUS
#=======================================================

echo
echo "======================================================="
echo " Minecraft 26.2 Hub installation completed"
echo "======================================================="
echo

echo "Java:"
java -version

echo

echo "Minecraft:"
echo "${MC_VERSION}"

echo

echo "Velocity:"
echo "${VELOCITY_VERSION}"

echo

echo "FabricProxy-Lite:"
echo "${FABRIC_PROXY_LITE_VERSION}"

echo

echo "ViaVersion:"
echo "${VIAVERSION_VERSION}"

echo

echo "Listening:"
ss -lntp || true

echo

echo "Firewall:"
firewall-cmd --list-all || true

echo

echo "Services:"
systemctl --no-pager --full status minecraft-hub.service || true

systemctl --no-pager --full status velocity.service || true

echo

echo "======================================================="
echo " Network"
echo "======================================================="
echo
echo "PUBLIC:"
echo "  TCP/25565 -> Velocity"
echo
echo "PRIVATE:"
echo "  127.0.0.1:25566 -> Fabric"
echo
echo "FORWARDING:"
echo "  Velocity modern forwarding"
echo "  FabricProxy-Lite"
echo
echo "SECRET:"
echo "  ${VELOCITY_ROOT}/forwarding.secret"
echo
echo "======================================================="


%end


#=======================================================
# REBOOT
#=======================================================

reboot
