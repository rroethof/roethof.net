### Select the smallest non-USB disk for both partitioning and GRUB.
# DESTRUCTIVE: inspect and test this logic against your hardware.
# Abort rather than silently continue if no eligible disk is found.
d-i partman/early_command string \
    set -eu; \
    best_disk=""; \
    best_size=""; \
    for disk in $(list-devices disk); do \
      name="$(basename "$disk")"; \
      sys_path="$(readlink -f "/sys/class/block/$name" 2>/dev/null || true)"; \
      case "$sys_path" in *"/usb"*) continue ;; *"/usb/"*) continue ;; esac; \
      sectors="$(cat "/sys/class/block/$name/size" 2>/dev/null || echo 0)"; \
      case "$sectors" in ''|*[!0-9]*) continue ;; esac; \
      [ "$sectors" -gt 0 ] || continue; \
      if [ -z "$best_size" ] || [ "$sectors" -lt "$best_size" ]; then best_disk="$disk"; best_size="$sectors"; fi; \
    done; \
    if [ -z "$best_disk" ]; then echo "ERROR: no eligible installation disk found" >/dev/tty; exit 1; fi; \
    debconf-set partman-auto/disk "$best_disk"; \
    debconf-set grub-installer/bootdev "$best_disk"
